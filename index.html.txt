'use client';

import React, { createContext, useContext, useEffect, useMemo, useState } from 'react';
import { PRODUCTS } from '@/data/catalog';
import { UI_COPY } from '@/data/translations';
import { CartLine, Locale, RouteView } from '@/types/ghosn';

interface AppContextValue {
  locale: Locale;
  toggleLocale: () => void;
  t: (typeof UI_COPY)[Locale];
  route: RouteView;
  selectedProductId: string;
  navigate: (view: RouteView, productId?: string) => void;
  cart: CartLine[];
  addToCart: (productId: string, sizeId: string, qty?: number) => void;
  updateQty: (productId: string, sizeId: string, qty: number) => void;
  removeLine: (productId: string, sizeId: string) => void;
  subtotal: number;
  totalItems: number;
  isCartOpen: boolean;
  setCartOpen: (open: boolean) => void;
  isSearchOpen: boolean;
  setSearchOpen: (open: boolean) => void;
  isAccountOpen: boolean;
  setAccountOpen: (open: boolean) => void;
}

const AppContext = createContext<AppContextValue | null>(null);

export function AppProvider({ children }: { children: React.ReactNode }) {
  const [locale, setLocale] = useState<Locale>('ar');
  const [localeReady, setLocaleReady] = useState(false);
  const [route, setRoute] = useState<RouteView>('home');
  const [selectedProductId, setSelectedProductId] = useState<string>(PRODUCTS[0].id);
  const [cart, setCart] = useState<CartLine[]>([]);
  const [isCartOpen, setCartOpen] = useState(false);
  const [isSearchOpen, setSearchOpen] = useState(false);
  const [isAccountOpen, setAccountOpen] = useState(false);

  useEffect(() => {
    // Restore the visitor's explicit language choice; Arabic remains the first-visit default.
    try {
      const savedLocale = window.localStorage.getItem('ghosn-locale');
      if (savedLocale === 'ar' || savedLocale === 'en') setLocale(savedLocale);
    } catch {
      // Storage can be unavailable in private browsing; the site still works without it.
    } finally {
      setLocaleReady(true);
    }
  }, []);

  useEffect(() => {
    const html = document.documentElement;
    html.lang = locale;
    html.dir = locale === 'ar' ? 'rtl' : 'ltr';
    if (!localeReady) return;
    try {
      window.localStorage.setItem('ghosn-locale', locale);
    } catch {
      // Language switching remains available for the current session.
    }
  }, [locale, localeReady]);

  const toggleLocale = () => {
    setLocale((prev) => (prev === 'en' ? 'ar' : 'en'));
  };

  const navigate = (view: RouteView, productId?: string) => {
    if (productId) setSelectedProductId(productId);
    setRoute(view);
    window.scrollTo({ top: 0, behavior: 'smooth' });
  };

  const addToCart = (productId: string, sizeId: string, qty = 1) => {
    setCart((prev) => {
      const idx = prev.findIndex((l) => l.productId === productId && l.sizeId === sizeId);
      if (idx > -1) {
        const copy = [...prev];
        copy[idx] = { ...copy[idx], quantity: copy[idx].quantity + qty };
        return copy;
      }
      return [...prev, { productId, sizeId, quantity: qty }];
    });
    setCartOpen(true);
  };

  const updateQty = (productId: string, sizeId: string, qty: number) => {
    if (qty <= 0) {
      removeLine(productId, sizeId);
      return;
    }
    setCart((prev) =>
      prev.map((l) =>
        l.productId === productId && l.sizeId === sizeId ? { ...l, quantity: qty } : l
      )
    );
  };

  const removeLine = (productId: string, sizeId: string) => {
    setCart((prev) => prev.filter((l) => !(l.productId === productId && l.sizeId === sizeId)));
  };

  const { subtotal, totalItems } = useMemo(() => {
    let sum = 0;
    let count = 0;
    for (const line of cart) {
      const product = PRODUCTS.find((p) => p.id === line.productId);
      const size = product?.sizes.find((s) => s.id === line.sizeId);
      if (size) sum += size.priceEGP * line.quantity;
      count += line.quantity;
    }
    return { subtotal: sum, totalItems: count };
  }, [cart]);

  return (
    <AppContext.Provider
      value={{
        locale,
        toggleLocale,
        t: UI_COPY[locale],
        route,
        selectedProductId,
        navigate,
        cart,
        addToCart,
        updateQty,
        removeLine,
        subtotal,
        totalItems,
        isCartOpen,
        setCartOpen,
        isSearchOpen,
        setSearchOpen,
        isAccountOpen,
        setAccountOpen,
      }}
    >
      {children}
    </AppContext.Provider>
  );
}

export function useApp() {
  const ctx = useContext(AppContext);
  if (!ctx) throw new Error('useApp must be used inside AppProvider');
  return ctx;
}

'use client';

import React, { useState } from 'react';
import { useApp } from '@/context/AppContext';
import { PRODUCTS } from '@/data/catalog';
import { submitOrderPreview } from '@/services/commerceAdapter';

export function CheckoutView() {
  const { locale, t, cart, subtotal, navigate } = useApp();
  const [form, setForm] = useState({
    fullName: '',
    email: '',
    phone: '',
    city: '',
    address: '',
    notes: '',
  });
  const [loading, setLoading] = useState(false);
  const [notice, setNotice] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);

  if (cart.length === 0) {
    return (
      <section className="reveal-view mx-auto max-w-2xl px-4 py-20 text-center">
        <h1 className="font-editorial text-4xl text-deep">{t.checkout.title}</h1>
        <p className="mt-4 text-sm text-deep/75">{t.checkout.emptyWarning}</p>
        <button
          onClick={() => navigate('shop')}
          className="mt-6 bg-olive px-7 py-3 text-xs uppercase tracking-widest text-cream hover:bg-deep"
        >
          {t.cart.emptyCta}
        </button>
      </section>
    );
  }

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setError(null);
    setNotice(null);

    if (!form.fullName || !form.email || !form.phone || !form.city || !form.address) {
      setError(t.checkout.requiredError);
      return;
    }

    setLoading(true);
    const lines = cart.map((line) => {
      const product = PRODUCTS.find((p) => p.id === line.productId)!;
      const size = product.sizes.find((s) => s.id === line.sizeId)!;
      return {
        sku: product.sku,
        sizeId: line.sizeId,
        quantity: line.quantity,
        unitPrice: size.priceEGP,
      };
    });

    const result = await submitOrderPreview({
      ...form,
      lines,
      subtotal,
      locale,
    });
    setLoading(false);
    setNotice(result.message[locale]);
  };

  return (
    <section className="reveal-view mx-auto max-w-6xl px-4 py-14 sm:px-8">
      <h1 className="font-editorial text-4xl text-deep">{t.checkout.title}</h1>
      <p className="mt-2 text-sm text-deep/75">{t.checkout.subtitle}</p>

      <div className="mt-10 grid gap-10 lg:grid-cols-12">
        <form onSubmit={handleSubmit} className="space-y-4 border border-sage/35 bg-paper p-7 lg:col-span-7">
          <h2 className="font-editorial text-2xl text-deep">{t.checkout.formTitle}</h2>

          <div className="grid gap-4 sm:grid-cols-2">
            <input
              type="text"
              aria-label={t.checkout.name}
              autoComplete="name"
              required
              placeholder={t.checkout.name}
              value={form.fullName}
              onChange={(e) => setForm({ ...form, fullName: e.target.value })}
              className="border border-sage/60 bg-cream px-3.5 py-2.5 text-sm focus:border-olive focus:outline-none"
            />
            <input
              type="email"
              aria-label={t.checkout.email}
              autoComplete="email"
              required
              placeholder={t.checkout.email}
              value={form.email}
              onChange={(e) => setForm({ ...form, email: e.target.value })}
              className="border border-sage/60 bg-cream px-3.5 py-2.5 text-sm focus:border-olive focus:outline-none"
            />
          </div>

          <div className="grid gap-4 sm:grid-cols-2">
            <input
              type="tel"
              aria-label={t.checkout.phone}
              autoComplete="tel"
              inputMode="tel"
              required
              placeholder={t.checkout.phone}
              value={form.phone}
              onChange={(e) => setForm({ ...form, phone: e.target.value })}
              className="border border-sage/60 bg-cream px-3.5 py-2.5 text-sm focus:border-olive focus:outline-none"
            />
            <input
              type="text"
              aria-label={t.checkout.city}
              autoComplete="address-level1"
              required
              placeholder={t.checkout.city}
              value={form.city}
              onChange={(e) => setForm({ ...form, city: e.target.value })}
              className="border border-sage/60 bg-cream px-3.5 py-2.5 text-sm focus:border-olive focus:outline-none"
            />
          </div>

          <input
            type="text"
            aria-label={t.checkout.address}
            autoComplete="street-address"
            required
            placeholder={t.checkout.address}
            value={form.address}
            onChange={(e) => setForm({ ...form, address: e.target.value })}
            className="w-full border border-sage/60 bg-cream px-3.5 py-2.5 text-sm focus:border-olive focus:outline-none"
          />

          <textarea
            rows={3}
            aria-label={t.checkout.notes}
            placeholder={t.checkout.notes}
            value={form.notes}
            onChange={(e) => setForm({ ...form, notes: e.target.value })}
            className="w-full border border-sage/60 bg-cream px-3.5 py-2.5 text-sm focus:border-olive focus:outline-none"
          />

          {error && <p className="text-xs text-red-700">{error}</p>}
          {notice && (
            <div className="border border-gold bg-cream p-4 text-sm text-deep">{notice}</div>
          )}

          <button
            type="submit"
            disabled={loading}
            className="w-full bg-olive py-3.5 text-xs font-semibold uppercase tracking-widest text-cream transition-colors hover:bg-deep disabled:opacity-60"
          >
            {loading ? t.checkout.submitting : t.checkout.submitBtn}
          </button>
        </form>

        <aside className="h-fit border border-sage/35 bg-paper p-7 lg:col-span-5">
          <h2 className="font-editorial text-2xl text-deep">{t.checkout.summaryTitle}</h2>
          <ul className="mt-5 space-y-3 border-b border-sage/30 pb-5 text-sm">
            {cart.map((line) => {
              const p = PRODUCTS.find((item) => item.id === line.productId);
              const s = p?.sizes.find((sz) => sz.id === line.sizeId);
              if (!p || !s) return null;
              return (
                <li key={`${line.productId}-${line.sizeId}`} className="flex justify-between gap-4">
                  <span className="text-deep/85">
                    {p.name[locale]} ({s.label[locale]}) × {line.quantity}
                  </span>
                  <span className="font-medium text-deep">
                    {s.priceEGP * line.quantity} {t.currency}
                  </span>
                </li>
              );
            })}
          </ul>
          <div className="mt-5 flex items-center justify-between">
            <span className="text-sm text-deep/80">{t.cart.subtotal}</span>
            <span className="font-editorial text-2xl font-medium text-deep">
              {subtotal} {t.currency}
            </span>
          </div>
          <p className="mt-2 text-xs text-deep/65">{t.cart.shippingNote}</p>
        </aside>
      </div>
    </section>
  );
}
